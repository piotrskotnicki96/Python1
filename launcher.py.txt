"""
P&S Automation - Launcher v0.4

Folder structure (the launcher creates missing folders itself):
    PS_Automation/
        launcher.py
        soi_pdf_parser.py
        soi_vanguard_parser.py        <- only for clients with "soi": {"layout": "vanguard"}
        matcher.py
        word_builder.py
        gdi_export.py                 <- GDI export (clients with a GDI template, e.g. Vanguard)
        clients/_template.json        <- full reference profile (not a client)
        clients/<client>/             <- EVERYTHING of one client in one folder:
            profile.json                 settings that differ from the defaults (SOI titles, name style, layout ...)
            mappings.xlsx                approved Asset ID -> name mappings
            funds.xlsx                   fund map: Fund Id -> SOI fund name, currency, order
            settings.json                last used paths, period and layout choice of this client
        clients/_default/             <- the same for the built-in "(default)" client (no profile.json)
        data/cache/                   <- slim copies of Bloomberg/PVAL files (safe to delete)
        output/<client>/<period>/

Older layouts (clients/<client>.json + data/mappings_<client>.xlsx + data/funds_<client>.xlsx) are migrated
automatically on start: the files are copied into the client's folder and the old ones renamed to *.migrated
(nothing is deleted).

Run:  python launcher.py
"""

# Folder with clients/, data/, output/. Built as .exe (PyInstaller) __file__ points into the program's
# internal/temp folder, so data would be lost or hidden there: use the folder of the .exe instead.
APP_DIR = Path(sys.executable).resolve().parent if getattr(sys, "frozen", False) else Path(__file__).resolve().parent
CLIENTS_DIR = APP_DIR / "clients"
OUTPUT_DIR = APP_DIR / "output"
SETTINGS_FILE = APP_DIR / "launcher_settings.json"
DATA_DIR = APP_DIR / "data"

# Client profile template. New client = copy of _template.json under a new name + ONLY the keys that differ.
# Matching/name settings come straight from matcher.DEFAULTS, so the launcher and matcher.py run alone
# always use the same rules (earlier the template kept its own, shorter copy of strip_suffixes).
TEMPLATE_PROFILE = {
    "client_name": "New client",
    "soi": {
        "titles": list(soi.SOI_TITLES),
        "ignore_contains": list(soi.IGNORE_CONTAINS),
        "layout": "standard",        # "vanguard" = one book PDF with many funds, 2-column equity pages
    },
    "alteryx": copy.deepcopy(matcher.DEFAULTS["alteryx"]),
    "name_format": copy.deepcopy(matcher.DEFAULTS["name_format"]),
    "fund_map": {},
    "matching": copy.deepcopy(matcher.DEFAULTS["matching"]),
    "output": copy.deepcopy(word_builder.OUTPUT_DEFAULTS),  # step 3 layout (Word/PDF)
}


DEFAULT_CLIENT = "(default)"
SETTINGS_KEYS = ["period", "sales_layout", "alteryx", "soi_cy", "soi_py", "bloomberg", "pval", "tj", "tj_on"]
MIGRATION_LOG = []


def client_dir(client):
    """Folder of one client; the built-in '(default)' client lives in clients/_default."""
    return CLIENTS_DIR / ("_default" if client in ("", DEFAULT_CLIENT) else client)


def migrate_legacy():
    """clients/<c>.json, data/mappings_<c>.xlsx, data/funds_<c>.xlsx -> clients/<c>/profile.json|mappings.xlsx|funds.xlsx.
    Copies, then renames the old file to *.migrated - nothing is deleted. Safe to run on every start."""
    log = MIGRATION_LOG
    for p in sorted(CLIENTS_DIR.glob("*.json")):
        if p.stem == "_template":
            continue
        d = CLIENTS_DIR / p.stem.strip()
        target = d / "profile.json"
        try:
            if not target.exists():
                d.mkdir(exist_ok=True)
                shutil.copy2(p, target)
            p.replace(p.with_name(p.name + ".migrated"))
            log.append(f"migrated profile {p.name} -> {p.stem}/profile.json")
        except OSError as e:
            log.append(f"WARNING: could not migrate {p.name} ({e})")
    for kind, new_name in (("mappings", "mappings.xlsx"), ("funds", "funds.xlsx")):
        for f in sorted(DATA_DIR.glob(f"{kind}_*.xlsx")):
            client = f.stem[len(kind) + 1:].strip()      # 'funds_Vanguard .xlsx' (stray space) -> 'Vanguard'
            if not client:
                continue
            d = CLIENTS_DIR / ("_default" if client == "default" else client)
            try:
                d.mkdir(exist_ok=True)
                if not (d / new_name).exists():
                    shutil.copy2(f, d / new_name)
                f.replace(f.with_name(f.name + ".migrated"))
                log.append(f"migrated {f.name} -> {d.name}/{new_name}")
            except OSError as e:
                log.append(f"WARNING: could not migrate {f.name} ({e})")


def ensure_structure():
    CLIENTS_DIR.mkdir(exist_ok=True)
    DATA_DIR.mkdir(exist_ok=True)
    OUTPUT_DIR.mkdir(exist_ok=True)
    migrate_legacy()
    tpl = CLIENTS_DIR / "_template.json"
    if not tpl.exists():
        tpl.write_text(json.dumps(TEMPLATE_PROFILE, indent=2, ensure_ascii=False), encoding="utf-8")


def load_clients():
    """'(default)' = built-in settings, no profile needed. Other entries = clients/<name>/profile.json
    (overrides only). A flat clients/<name>.json that has not been migrated yet is still understood."""
    out = {DEFAULT_CLIENT: None}
    for d in sorted(CLIENTS_DIR.iterdir()):
        if d.is_dir() and not d.name.startswith("_") and (d / "profile.json").exists():
            out[d.name] = d / "profile.json"
    for p in sorted(CLIENTS_DIR.glob("*.json")):
        if p.stem != "_template" and p.stem not in out:
            out[p.stem] = p
    return out


def load_client_settings(client):
    p = client_dir(client) / "settings.json"
    if not p.exists():
        return {}
    try:
        return json.loads(p.read_text(encoding="utf-8"))
    except (OSError, ValueError):
        return {}


def save_client_settings(client, data):
    """Last used paths/period of ONE client. Returns an error message, or None."""
    try:
        d = client_dir(client)
        d.mkdir(parents=True, exist_ok=True)
        (d / "settings.json").write_text(
            json.dumps({k: data.get(k, "") for k in SETTINGS_KEYS}, indent=2, ensure_ascii=False), encoding="utf-8")
        return None
    except OSError as e:
        return f"could not save the settings of client '{client}' ({e})"


def deep_merge(base, override):
    for k, v in (override or {}).items():
        if isinstance(v, dict) and isinstance(base.get(k), dict):
            deep_merge(base[k], v)
        else:
            base[k] = v
    return base


def load_settings():
    """Last used paths. Missing file = first run; a damaged file is ignored (only convenience data)."""
    if not SETTINGS_FILE.exists():
        return {}
    try:
        return json.loads(SETTINGS_FILE.read_text(encoding="utf-8"))
    except (OSError, ValueError):
        return {}


def save_settings(data):
    """Returns an error message, or None when saved."""
    try:
        SETTINGS_FILE.write_text(json.dumps(data, indent=2, ensure_ascii=False), encoding="utf-8")
        return None
    except OSError as e:
        return f"could not save launcher settings ({e})"

    # ---------- per-client settings ----------
    def _fields_snapshot(self):
        return {"period": self.period_var.get(), "sales_layout": self.sales_layout.get(),
                **{k: v.get() for k, v in self.vars.items()}}

    def _store_client_settings(self):
        """Saves the fields under the client they were filled in for (not the one just selected)."""
        if self._active_client is None:
            return
        err = save_client_settings(self._active_client, self._fields_snapshot())
        if err:
            self._log(f"WARNING: {err}")

    def _load_client_into_fields(self, client):
        st = load_client_settings(client)
        if not st and client == self.settings.get("client"):      # first start after the upgrade: old global values
            st = {k: self.settings.get(k, "") for k in SETTINGS_KEYS}
        self.period_var.set(st.get("period", ""))
        self.sales_layout.set(st.get("sales_layout", "(profile)") or "(profile)")
        for k, v in self.vars.items():
            v.set(st.get(k, ""))
        self._active_client = client

    def _on_client_changed(self):
        self._store_client_settings()
        self._load_client_into_fields(self.client_var.get())
        self._log(f"Client: {self.client_var.get()} -> {client_dir(self.client_var.get())}")

    def _on_close(self):
        self._store_client_settings()
        save_settings({"client": self.client_var.get()})
        self.destroy()

    def _on_new_client(self):
        name = simpledialog.askstring("New client", "Client name (becomes the folder name):", parent=self)
        name = "".join(ch for ch in (name or "") if ch.isalnum() or ch in " _-").strip()
        if not name or name.startswith("_") or name == DEFAULT_CLIENT:
            self._log("New client cancelled (no valid name).")
            return
        d = CLIENTS_DIR / name
        if (d / "profile.json").exists():
            self._log(f"Client '{name}' already exists - selected.")
        else:
            cur = self.client_var.get()
            src = self.clients.get(cur)
            data = {"client_name": name}
            if src and messagebox.askyesno("New client", f"Copy the profile of '{cur}' as the starting point?\n"
                                           "(No = empty profile with built-in settings.)", parent=self):
                try:
                    data = json.loads(src.read_text(encoding="utf-8"))
                    data["client_name"] = name
                except (OSError, ValueError) as e:
                    self._log(f"WARNING: could not copy the profile of '{cur}' ({e}) - empty profile created.")
            d.mkdir(parents=True, exist_ok=True)
            (d / "profile.json").write_text(json.dumps(data, indent=2, ensure_ascii=False), encoding="utf-8")
            self._log(f"Created {d / 'profile.json'}")
        self._store_client_settings()
        self.clients = load_clients()
        self.client_cb["values"] = list(self.clients)
        self.client_var.set(name)
        self._load_client_into_fields(name)

    def _open_client_folder(self):
        import os
        d = client_dir(self.client_var.get())
        d.mkdir(parents=True, exist_ok=True)
        os.startfile(d) if hasattr(os, "startfile") else self._log(str(d))

    def _browse(self, key, kind):
        if kind == "dir":
            p = filedialog.askdirectory()
        else:
            ft = [("PDF", "*.pdf")] if kind == "pdf" else [("Excel / CSV", "*.xlsx *.xls *.csv")]
            p = filedialog.askopenfilename(filetypes=ft + [("All files", "*.*")])
        if p:
            self.vars[key].set(p)

    def _refresh_clients(self):
        self.clients = load_clients()
        self.client_cb["values"] = list(self.clients)
        self._log(f"Profiles: {', '.join(self.clients)}")

    def _log(self, msg):
        def _w():
            self.log.configure(state="normal")
            self.log.insert("end", f"[{datetime.now():%H:%M:%S}] ", "ts")
            low = str(msg).lstrip().lower()
            tag = "err" if low.startswith(("error", "traceback")) else "warn" if low.startswith("warning") \
                else "ok" if ("done" in low[:40]) else ()
            self.log.insert("end", f"{msg}\n", tag)
            self.log.see("end")
            self.log.configure(state="disabled")
        self.after(0, _w)

    def _open_output(self):
        import os
        path = self._out_dir()
        path.mkdir(parents=True, exist_ok=True)
        os.startfile(path) if hasattr(os, "startfile") else self._log(str(path))

    # ---------- logic ----------
    def _snapshot(self):
        """All field values, read ONCE on the main thread when a step starts. The worker thread uses only this
        copy, so changing the client/period/paths while a step runs cannot mix two clients' inputs."""
        return {"client": self.client_var.get(), "period": self.period_var.get(),
                "sales_layout": self.sales_layout.get(),
                **{k: v.get() for k, v in self.vars.items()}}

    def _profile(self, snap):
        """Built-in settings + (optionally) the client's JSON on top - the JSON may contain only what differs."""
        prof = json.loads(json.dumps(TEMPLATE_PROFILE))
        path = self.clients.get(snap["client"])
        self._log(f"Client profile: {snap['client']} -> {path if path else 'built-in settings (no JSON)'}")
        if path:
            try:
                deep_merge(prof, json.loads(path.read_text(encoding="utf-8")))
            except ValueError as e:
                raise ValueError(f"The profile file has a JSON syntax error: {e}\nFile: {path}\n"
                                 "Typical cause: a missing comma at the end of the previous line, or a missing quote/bracket.") from None
        if snap.get("sales_layout") in ("always", "auto", "fit"):      # dropdown overrides the profile
            prof.setdefault("output", {})["sales_new_page"] = snap["sales_layout"]
        return prof

    def _out_dir(self, snap=None):
        snap = snap or self._snapshot()
        period = snap["period"].replace("/", "-").replace("\\", "-").replace("..", "") or "no_period"
        client = snap["client"]
        client = "default" if client in ("", DEFAULT_CLIENT) else client
        return OUTPUT_DIR / client / period

    def _run(self, fn):
        snap = self._snapshot()
        self._store_client_settings()
        err = save_settings({"client": snap["client"]})
        if err:
            self._log(f"WARNING: {err}")
        btns = (self.b1, self.b2, self.b4, self.b5, self.b3, self.b8, self.b6, self.b7)
        for b in btns:
            b.configure(state="disabled")
        self.progress.start(12)

        def worker():
            try:
                fn(snap)
            except Exception as e:
                self._log(f"ERROR: {e}")
                self._log(traceback.format_exc())
            finally:
                self.after(0, lambda: ([b.configure(state="normal") for b in btns], self.progress.stop()))
        threading.Thread(target=worker, daemon=True).start()

    def _parse_soi(self, path, prof):
        """Parsed SOI with a disk cache: the same PDF(s) + same SOI settings + same parser code = the saved result
        (step 2 no longer re-reads a 1000+ page book that step 1 already parsed)."""
        import hashlib
        import pickle
        import time
        cfg = prof["soi"]
        try:
            p = Path(path)
            files = [p] if p.is_file() else sorted(f for f in p.rglob("*") if f.is_file() and f.suffix.lower() == ".pdf")
            parts = [(str(f.resolve()), f.stat().st_size, int(f.stat().st_mtime)) for f in files]
            code = [(m.__file__, int(Path(m.__file__).stat().st_mtime)) for m in (soi, soi_vg)]
            key = hashlib.md5(json.dumps([parts, code, cfg], sort_keys=True, default=str).encode("utf-8")).hexdigest()
            cache = DATA_DIR / "cache" / f"soi_parsed_{key}.pkl"
        except (OSError, TypeError, ValueError):
            cache = None
        if cache is not None and cache.exists():
            try:
                with open(cache, "rb") as fh:
                    res = pickle.load(fh)
                self._log(f"  parsed SOI loaded from cache ({Path(path).name})")
                return res
            except Exception:  # noqa: BLE001 - damaged cache = parse again
                pass
        t0 = time.time()
        res = self._parse_soi_raw(path, prof)
        self._log(f"  SOI parsed in {time.time() - t0:.0f}s")
        if cache is not None:
            try:
                cache.parent.mkdir(parents=True, exist_ok=True)
                for old in cache.parent.glob("soi_parsed_*.pkl"):      # keep the cache folder small
                    if time.time() - old.stat().st_mtime > 14 * 86400:
                        old.unlink(missing_ok=True)
                with open(cache, "wb") as fh:
                    pickle.dump(res, fh)
            except Exception:  # noqa: BLE001 - cache is optional
                pass
        return res

    def _parse_soi_raw(self, path, prof):
        """Standard parser, or the Vanguard book parser when the client profile says so."""
        cfg = prof["soi"]
        if cfg.get("layout") == "vanguard":
            return soi_vg.parse_path(path, cfg.get("titles"), cfg.get("ignore_contains"), log=self._log,
                                     funds=cfg.get("funds") or None, cache_dir=DATA_DIR / "cache")
        return soi.parse_path(path, cfg.get("titles"), cfg.get("ignore_contains"), log=self._log)

    def step_soi(self, snap):
        prof = self._profile(snap)
        out = self._out_dir(snap)
        out.mkdir(parents=True, exist_ok=True)
        for key, label in (("soi_cy", "SOI_CY"), ("soi_py", "SOI_PY")):
            path = snap[key]
            if not path:
                self._log(f"{label}: skipped (no file)")
                continue
            self._log(f"{label}: reading {Path(path).name} ...")
            df, sus, chk = self._parse_soi(path, prof)
            target = out / f"{label}_parsed.xlsx"
            soi.save_excel(df, sus, target, chk)
            bad = int((chk["Status"] != "OK").sum()) if not chk.empty else 0
            self._log(f"{label}: sector subtotal check - {len(chk)} checked, {bad} with difference")
            funds = df["Fund"].nunique() if not df.empty else 0
            self._log(f"{label}: securities {len(df)} | funds {funds} | to review {len(sus)}")
            self._log(f"{label}: saved {target}")
        self._log("Step 1 done.")

    def step_gdi(self, snap, review_path):
        if not snap["period"].strip():
            raise ValueError("Enter the Period (dd/mm/yyyy) - it is the snapshot date in the GDI file.")
        prof = self._profile(snap)
        gcfg = prof.get("gdi") or {}
        tpl = Path(gcfg.get("template") or "GDI_template.xlsx")
        if not tpl.is_absolute():
            tpl = client_dir(snap["client"]) / tpl
        tpl = gdi_export.find_template(tpl)
        out = self._out_dir(snap)
        out.mkdir(parents=True, exist_ok=True)
        target = out / f"GDI_{datetime.now():%Y%m%d_%H%M}{'.xlsm' if tpl.suffix.lower() in ('.xlsm', '.xltm') else '.xlsx'}"
        self._log(f"Building GDI from {Path(review_path).name} using template {tpl} ...")
        files = gdi_export.export_gdi(review_path, tpl, target, snap["period"], gcfg, log=self._log)
        self._log(f"Step 3b done. {len(files)} GDI file(s) in {out}. Check a few rows against the Review before loading to GDI.")

    def step_import_pdf(self, pdf_path, client):
        diff, rows = profile_import.extract_output_from_pdf(pdf_path)
        self._log(f"Layout measured on {Path(pdf_path).name}:")
        self._log(profile_import.format_report(rows))
        if not diff:
            self._log("Nothing to save (no differences from the defaults, or the PDF does not look like the note).")
            return
        self.after(0, lambda: self._confirm_save_profile(diff, client))

    def step_import_word(self, docx_path, client):
        diff, rows = profile_import.extract_output_settings(docx_path)
        self._log(f"Layout read from {Path(docx_path).name}:")
        self._log(profile_import.format_report(rows))
        try:
            self._log("Structure of the Word file (send this when the layout is read wrongly):\n"
                      + profile_import.describe_structure(docx_path))
        except Exception as e:                       # the dump is only a diagnostic aid
            self._log(f"(structure dump failed: {e})")
        if not diff:
            self._log("Nothing to save (no differences from the defaults, or the file does not look like the note).")
            return
        self.after(0, lambda: self._confirm_save_profile(diff, client))

    def _confirm_save_profile(self, diff, client):
        """Runs on the main thread: the user sees what would be written and decides."""
        lines = "\n".join(f"{k}: {str(v)[:60]}" for k, v in diff.items())
        if messagebox.askyesno("Save layout to profile",
                               f"Save {len(diff)} setting(s) to clients/{client}/profile.json?\n"
                               f"(an existing profile is backed up as profile.json.bak)\n\n{lines}", parent=self):
            p = profile_import.save_client_profile(diff, client, CLIENTS_DIR)
            self._log(f"Saved {len(diff)} setting(s) -> {p}")
            self._refresh_clients()
            self._log(f"Pick client '{client}' in the list and generate a test note.")
        else:
            self._log("Layout NOT saved.")

    def step_word(self, snap, review_path):
        if not snap["period"].strip():
            raise ValueError("Enter the Period (dd/mm/yyyy) - it is printed in the note header.")
        out = self._out_dir(snap)
        target = out / f"PS_Note_{datetime.now():%Y%m%d_%H%M}.docx"
        self._log(f"Building Word from {Path(review_path).name} ...")
        prof = self._profile(snap)
        src = self.clients.get(snap["client"])
        self._log(f"Profile: {snap['client']} ({(src.parent.name + '/' + src.name) if src else 'built-in settings, no JSON file'}) | "
                  f"page_numbers={prof['output'].get('page_numbers')} | first_page_number={prof['output'].get('first_page_number')}")
        word_builder.build_note(review_path, target, prof, snap["period"],
                                                   fund_map_path=self._fund_map_file(snap), log=self._log)
        self._log("Step 3 done. Check the Word_Check file and read the document before publishing.")

    def step_match(self, snap):
        prof = self._profile(snap)
        out = self._out_dir(snap)
        out.mkdir(parents=True, exist_ok=True)
        folder = snap["alteryx"]
        if not folder:
            raise ValueError("Select the folder with Alteryx files.")
        dfs = {}
        for key, label in (("soi_cy", "SOI_CY"), ("soi_py", "SOI_PY")):
            path = snap[key]
            if path:
                self._log(f"{label}: reading {Path(path).name} ...")
                dfs[key] = self._parse_soi(path, prof)[0]
                self._log(f"{label}: {len(dfs[key])} securities")
            else:
                dfs[key] = None
                self._log(f"{label}: no file - skipped")
        externals = {label: snap[key] for key, label in (("bloomberg", "BLOOMBERG"), ("pval", "PVAL"))
                     if snap[key]}
        target = out / f"Review_{datetime.now():%Y%m%d_%H%M}.xlsx"
        matcher.run_matching(folder, dfs["soi_cy"], dfs["soi_py"], prof, target,
                             mappings_path=self._mappings_file(snap), log=self._log,
                             fund_map_path=self._fund_map_file(snap), external_paths=externals,
                             tj_path=(snap.get("tj") or None) if snap.get("tj_on") == "1" else None,
                             cache_dir=APP_DIR / "data" / "cache")
        self._log("Step 2 done. Open the Review file and check the yellow and red rows.")
