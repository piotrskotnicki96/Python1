"""Numeric layout comparison of two P&S note PDFs (ours vs the original publication).

    python pdf_compare.py ours.pdf original.pdf            (or run without arguments -> file dialogs)

Per PDF it measures, on the pages that contain a Purchases/Sales table: page size, where the table header starts
(distance from the top), how far the last row is from the bottom edge, line pitch (pt), rows on the fullest page, left/right
edge of the text and the font size. Differences between the two columns tell which setting to change
(bottom margin, line_pt, font_size, top margin, col widths ...)."""
import statistics
import sys
from pathlib import Path

import pdfplumber

CM = 28.3465


def _size_name(w, h):
    for name, (a, b) in {"A4": (21.0, 29.7), "Letter": (21.59, 27.94), "Legal": (21.59, 35.56)}.items():
        if abs(w / CM - a) < 0.25 and abs(h / CM - b) < 0.25:
            return name
    return "other"


def measure(pdf_path, max_pages=60):
    pages = []
    with pdfplumber.open(str(pdf_path)) as pdf:
        n_total = len(pdf.pages)
        for pg in pdf.pages[:max_pages]:
            words = pg.extract_words(extra_attrs=["size"])
            heads = [w for w in words if w["text"].lower() in ("major", "total") and any(
                v["text"].lower() in ("purchases", "sales") and abs(v["top"] - w["top"]) < 2 and 0 < v["x0"] - w["x1"] < 12
                for v in words)]
            if not heads:
                continue
            h0 = min(h["top"] for h in heads)
            body = [w for w in words if w["top"] >= h0 - 1]
            lines = {}
            for w in body:
                lines.setdefault(round(w["top"]), []).append(w)
            ks = sorted(lines)
            if len(ks) > 3 and len(lines[ks[-1]]) == 1 and lines[ks[-1]][0]["text"].isdigit():
                ks = ks[:-1]            # page number
            if len(ks) < 2:
                continue
            tops = [min(w["top"] for w in lines[k]) for k in ks]
            pitch = statistics.median(b - a for a, b in zip(tops, tops[1:]) if b - a > 3) if len(tops) > 2 else None
            ws = [w for k in ks for w in lines[k]]
            pages.append({"w": pg.width, "h": pg.height, "head_top": h0 / CM,
                          "bottom_gap": (pg.height - max(w["bottom"] for w in ws)) / CM,
                          "lines": len(ks), "pitch": pitch, "left": min(w["x0"] for w in ws) / CM,
                          "right": (pg.width - max(w["x1"] for w in ws)) / CM,
                          "size": statistics.median(w["size"] for w in ws)})
    return n_total, pages


def summarize(path):
    n, pages = measure(path)
    if not pages:
        return {"file": Path(path).name, "pages": n, "error": "no Purchases/Sales table pages found"}
    full = max(pages, key=lambda p: p["lines"])
    cont = [p for p in pages if p["head_top"] < 6] or pages
    return {"file": Path(path).name, "pages": n,
            "page size": f"{pages[0]['w'] / CM:.2f} x {pages[0]['h'] / CM:.2f} cm ({_size_name(pages[0]['w'], pages[0]['h'])})",
            "table header from top (cm)": f"{statistics.median(p['head_top'] for p in cont):.2f}",
            "rows on the fullest page": full["lines"],
            "gap below last row, fullest page (cm)": f"{full['bottom_gap']:.2f}",
            "line pitch (pt)": f"{full['pitch']:.2f}" if full["pitch"] else "-",
            "font size (pt)": f"{full['size']:.1f}",
            "text left edge (cm)": f"{min(p['left'] for p in pages):.2f}",
            "text right edge (cm)": f"{min(p['right'] for p in pages):.2f}"}


def main():
    args = [a for a in sys.argv[1:] if not a.startswith("-")]
    if len(args) < 2:
        import tkinter as tk
        from tkinter import filedialog
        tk.Tk().withdraw()
        args = [filedialog.askopenfilename(title="OUR PDF"), filedialog.askopenfilename(title="ORIGINAL PDF")]
    a, b = summarize(args[0]), summarize(args[1])
    keys = [k for k in a if k != "file"]
    print(f"{'':<42}{'OURS':<30}{'ORIGINAL':<30}")
    print(f"{'file':<42}{a['file'][:28]:<30}{b['file'][:28]:<30}")
    for k in keys:
        print(f"{k:<42}{str(a.get(k, '-')):<30}{str(b.get(k, '-')):<30}")


if __name__ == "__main__":
    main()
