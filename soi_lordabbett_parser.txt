"""
Lord Abbett SOI parser v1.0 - Schedule of Investments of Lord Abbett Global Funds I plc (annual / interim report,
one PDF for all funds). Used ONLY for clients with "soi": {"layout": "lord_abbett"} in profile.json; every other
client keeps soi_pdf_parser.py (this file is a copy of soi_pdf_parser v1.3 + Lord Abbett rules, changes here
never touch other clients).

Returns the SAME tables as soi_pdf_parser.parse_pdf (securities, suspicious lines, sector checks), so the
matcher / Review / Word work unchanged. Extra column in SOI: Note.

Usage (standalone test):
    python soi_lordabbett_parser.py report.pdf   -> <pdf_name>_SOI_parsed.xlsx next to the PDF

Lord Abbett specifics handled:
    - perpetuals printed with a dash instead of a date ('7.00% (2) - (3)')        -> Note = PERPETUAL
    - mortgage TBAs printed with 'TBA' instead of a date ('5.50% TBA')            -> Note = TBA
    - term loans with coupon 'TBD'                                                -> Note = COUPON TBD
    - dated securities without a coupon ('ACBL Holding Corp.* 01/04/2045 ...')    -> Note = NO COUPON
    - '*' (non-income producing) after a name goes to Footnotes; fair value '–' = 0
    - derivative blocks (forwards, futures, swaps, options) are skipped completely
    - '(continued)' sector / section headings on the next page do not reset the subtotal check;
      every heading followed by italic sub-sectors becomes the section (Asset-Backed / Corporate Bonds / ...)
    - 'Total Investment in Securities (Cost: ...)' is not read as an equity
    - single-digit principal ('15/02/2030 3 2,961 0.00%') is the holding, not a footnote
    - 'Principal (000)' -> holdings x 1000

Tested: FY 31/01/2026 annual report, 12 funds - 5,058 securities, To Review 0, sector subtotals 529/529 OK.

Requires: pip install pdfplumber pandas openpyxl
"""

import re
import sys
from datetime import date
from pathlib import Path

import pandas as pd
import pdfplumber

# ============================ CONFIG ============================
# Defaults only - never modified at runtime. A client profile passes its own values into parse_pdf().
# SOI page titles (case-insensitive). Pages without any of them are skipped, unless they are
# continuation pages of an SOI (see parse_pdf).
SOI_TITLES = ["STATEMENT OF INVESTMENTS", "SCHEDULE OF INVESTMENTS", "PORTFOLIO OF INVESTMENTS"]
# Lines to ignore (table headers, footers)
# Table header words: a line is a header only if it contains 2+ of them (or is exactly one of them),
# so names like "Alibaba Group Holding Ltd" are NOT dropped.
HEADER_WORDS = ["Security Description", "Fair Value", "Holding", "% of Sub-Fund", "% of Net Assets", "Nominal"]
# Phrases that are never security names (totals/footers) - substring match.
IGNORE_CONTAINS = ["Net Current Assets", "Financial Assets at", "Financial Liabilities at", "Cash at Bank",
                   "Other Net Assets", "Net Assets attributable", "Total Investment"]
# Headings that mean a page belongs to another statement (used for continuation pages without a title)
# (no bare "REPORT": running headers such as "Annual Report 2025" are on every page)
OTHER_HEADINGS = ["STATEMENT OF", "NOTES TO", "SCHEDULE OF", "PORTFOLIO OF", "REPORT OF", "AUDITOR", "BALANCE SHEET"]
# The SOI title must be in the first lines of a page (a notes page may mention "Schedule of Investments" in prose)
TITLE_TOP_LINES = 8
# Maximum number of lines a wrapped security name can span
MAX_WRAP_LINES = 3
# A sector/section heading containing one of these starts a derivative block (forwards, futures, swaps ...):
# everything until the next non-derivative heading or the next fund is skipped (not part of P&S).
DERIVATIVE_HEADINGS = re.compile(r"\b(Forward|Futures?|Swaps?|Options?|Swaptions?)\b", re.I)
# ================================================================

# Debt security: "Meituan, 5.125%, 05/11/35 ..." or "United States Treasury Bill ZCP, 29/04/25 ..."
# Coupon: 5.125% / 5.1% / 5% / ZCP / TBD. Date: dd/mm/yy, d/m/yy, dd/mm/yyyy, "Perpetual",
# "-" (perpetual, Lord Abbett) or "TBA" (mortgage to-be-announced, Lord Abbett).
# A 4-digit year only when no digit/comma follows: long names can glue the date to the holding
# ("15/05/30500,000" = 15/05/30 + 500,000).
# Also read: no commas ("Name 3.574% 15/02/2050 11,040 ...") and footnote markers glued to the coupon ("3.526%(1)").
SEC_RE = re.compile(
    r"^(?P<name>.+?),?\s+(?P<cpn>\d{1,2}(?:\.\d{1,4})?%|ZCP|TBD)(?P<cfn>(?:\s*\(\s*\w{1,3}\s*\))*),?\s*"
    r"(?P<date>\d{1,2}/\d{1,2}/(?:\d{4}(?![\d,])|\d{2})|Perpetual\b|TBA\b|[-–](?=\s|\(|$))(?P<rest>.*)$", re.I
)
# Dated security WITHOUT a coupon: "ACBL Holding Corp.* 01/04/2045 2,494 116,595 0.01%" (name + date + holding + value + %)
NOCPN_RE = re.compile(
    r"^(?P<name>.*[A-Za-z].*?)\s+(?P<date>\d{1,2}/\d{1,2}/\d{4})\s+(?:[A-Z]{3}\s+)?(?P<hold>\(?[\d,]+(?:\.\d+)?\)?)\s+"
    r"(?P<fv>\(?[\d,]+(?:\.\d+)?\)?|[–-]\s*(?:\(\s*\w{1,3}\s*\))?)\s+(?P<pct>\(?-?[\d.]+%\)?)\s*$"
)
# Country/sector header: "Financial - 3.74% (31st December, 2024: 7.22%)" / "Open Forward ... - (0.00%) (31 January 2025: ...)"
HEADER_RE = re.compile(r"^(?P<label>.+?)\s+[-–]\s+\(?[\d.,–-]*%?\)?\s*\(\s*\d{1,2}(?:st|nd|rd|th)?\s+[A-Za-z]", re.I)
DATE_ANYWHERE_RE = re.compile(r"\d{1,2}/\d{1,2}/\d{2,4}")
NUM_RE = re.compile(r"^\(?-?[\d,]+(\.\d+)?\)?%?$")
# Equity: "State Bank of India 4,053,980 38,337,394 3.52%" (name + holding + fair value + %)
# Fair value '–' (Lord Abbett, worthless position), optionally with a glued footnote: "–(6)".
EQ_RE = re.compile(
    r"^(?P<name>.*[A-Za-z].*?)\s+(?P<hold>\(?[\d,]+(?:\.\d+)?\)?)\s+"
    r"(?P<fv>\(?[\d,]+(?:\.\d+)?\)?|[–-]\s*(?:\(\s*\w{1,3}\s*\))?)\s+(?P<pct>\(?-?[\d.]+%\)?)\s*$"
)
# Footnote markers at the end of an equity name: (1), (a), (ii), * (non-income producing).
# NOT share-class tags like (Pref), (ADR), (DVR).
TRAIL_FOOTNOTE_RE = re.compile(r"\s*(?:\(\s*(?:\d{1,2}|[a-z]|[ivx]{1,4})\s*\)|\*+)\s*$")
# Sector subtotal: "115,997,163 8.87%"
SUBTOTAL_RE = re.compile(r"^\(?[\d,]+\)?\s+\(?-?[\d.]+%\)?$")
# FX forwards: "JPY 1,243,864,282 USD (8,197,669) ..." - not part of P&S
FORWARD_RE = re.compile(r"^[A-Z]{3}\s+\(?[\d,]+\)?\s+[A-Z]{3}\s+\(?[\d,]+\)?")
FOOTNOTE_RE = re.compile(r"^\s*(\(\s*[\w]{1,4}\s*\)|\d(?=\s|$))")
CONTINUED_RE = re.compile(r"\s*\(continued\)\s*$", re.I)


def to_number(tok):
    tok = tok.replace(",", "").replace("%", "")
    neg = tok.startswith("(") and tok.endswith(")")
    tok = tok.strip("()")
    try:
        v = float(tok)
    except ValueError:
        return None
    return -v if neg else v


def fv_number(tok):
    """Fair value token: '1,234' / '(1,234)' / '–' / '–(6)' (dash = nil, Lord Abbett)."""
    t = tok.strip()
    if t[:1] in "–-" and not re.match(r"-\d", t):
        return 0.0
    return to_number(t)


def dash_footnote(tok):
    """'–(6)' -> '(6)' (footnote glued to a dash fair value), otherwise ''."""
    m = re.search(r"\(\s*\w{1,3}\s*\)", tok) if tok.strip()[:1] in "–-" else None
    return m.group(0).replace(" ", "") if m else ""


def parse_date(text):
    """'05/11/35' -> 2035-11-05 (2-digit year always 20xx: perpetuals 73/74 -> 2073/2074);
    '05/11/2035' -> 2035-11-05; 'Perpetual' / '-' / 'TBA' -> None."""
    if "/" not in text:
        return None
    d, m, y = (int(x) for x in text.split("/"))
    try:
        return date(y if y >= 100 else 2000 + y, m, d)
    except ValueError:
        return None


def date_note(date_txt, cpn_txt):
    """Note column for securities without a normal coupon/date."""
    notes = []
    d = date_txt.strip().upper()
    if d in ("-", "–", "PERPETUAL"):
        notes.append("PERPETUAL")
    elif d == "TBA":
        notes.append("TBA")
    if cpn_txt.upper() == "TBD":
        notes.append("COUPON TBD")
    return ", ".join(notes)


def strip_trailing_footnotes(name):
    """'Vistra Corp.(1)' -> ('Vistra Corp.', ['(1)']); 'DexCom, Inc.*' -> ('DexCom, Inc.', ['*'])."""
    fns = []
    while TRAIL_FOOTNOTE_RE.search(name):
        fn = TRAIL_FOOTNOTE_RE.search(name)
        fns.insert(0, fn.group(0).strip())
        name = name[:fn.start()].strip()
    return name, fns


def is_noise(line, ignore=IGNORE_CONTAINS, header_words=HEADER_WORDS):
    s = line.strip()
    if not s:
        return True
    low = s.lower()
    hw = [w for w in header_words if w.lower() in low]
    if len(hw) >= 2 or low in {w.lower() for w in header_words}:
        return True
    if any(k.lower() in low for k in ignore):
        return True
    # "Total Investments 1,234 98.5%" is noise; "TotalEnergies SE" / "Total Access Communication 1,000 2,000 0.5%"
    # (name + holding + value + %) is a security
    if re.match(r"TOTAL\b", s.upper()) and not EQ_RE.match(s):
        return True
    if FORWARD_RE.match(s):
        return True
    if all(NUM_RE.match(t) for t in s.split()):  # numbers only (subtotals)
        return True
    # NOTE: ALL CAPS lines are NOT treated as noise any more - some clients print security names in
    # capitals. Page headers/countries are harmless: the buffer is reset by the table header line.
    return False


TRAILING_DESIG_RE = re.compile(r"^\s*(144A)(?=[\W\d]|$)", re.I)


def split_rest(rest):
    """Separates designations kept in the name ('144A' after the date) and footnotes (FRN)/(1)
    from the numbers: holding, fair value, %."""
    footnotes, desig = [], []
    r = rest
    while True:
        d = TRAILING_DESIG_RE.match(r)
        if d:
            desig.append(d.group(1).upper())
            r = r[d.end():]
            continue
        m = FOOTNOTE_RE.match(r)
        if not m:
            break
        # a bare digit is a footnote only if holding + value + % still follow it;
        # otherwise it IS the holding ('4.875% 15/02/2030 3 2,961 0.00%' = principal 3 (000))
        if not m.group(1).startswith("(") and sum(1 for t in r[m.end():].split() if NUM_RE.match(t)) < 3:
            break
        footnotes.append(m.group(1).strip())
        r = r[m.end():]
    toks = r.split()
    nums = [t for t in toks if NUM_RE.match(t)]
    pct = next((to_number(t) for t in nums if t.endswith("%")), None)
    plain = [to_number(t) for t in nums if not t.endswith("%")]
    holding = plain[0] if len(plain) > 0 else None
    fair_value = plain[1] if len(plain) > 1 else None
    if fair_value is None and holding is not None:
        # dash fair value '–' / '–(3)' after the holding (worthless position)
        dash = next((t for t in toks if t[:1] in "–-" and not re.match(r"-\d", t)), None)
        if dash is not None and pct is not None:
            fair_value = 0.0
            fn = dash_footnote(dash)
            if fn:
                footnotes.append(fn)
    return footnotes, holding, fair_value, pct, desig


def has_title(lines, titles):
    """SOI title within the first TITLE_TOP_LINES non-empty lines of the page."""
    top = [x for x in lines if x.strip()][:TITLE_TOP_LINES]
    return any(t in x.upper() for x in top for t in titles)


def looks_like_name_part(text):
    """First line of a security name wrapped over two lines: words, no numbers or %, not an ALL-CAPS label."""
    t = text.strip()
    return bool(t) and not re.search(r"%|\d,\d{3}|\d/\d", t) and not t.isupper() and not t.endswith(":") \
        and not all(NUM_RE.match(x) for x in t.split())


def italic_line_texts(page):
    """Texts of the lines printed entirely in italics (sub-sector headings such as 'Commercial Mortgage-Backed',
    'Credit Card'). Used so that such a heading is not glued to the first security below it."""
    out = set()
    try:
        for ln in page.extract_text_lines():
            letters = [c for c in ln["chars"] if c["text"].strip() and c["text"].isalpha()]
            if letters and all(re.search(r"italic|oblique", c.get("fontname", ""), re.I) for c in letters):
                out.add(ln["text"].strip())
    except Exception:  # noqa: BLE001 - optional helper
        pass
    return out


def find_fund(lines, titles):
    for i, ln in enumerate(lines):
        if any(t in ln.upper() for t in titles):
            for j in range(i - 1, -1, -1):
                if lines[j].strip():
                    return lines[j].strip()
            return None
    return None


def parse_pdf(pdf_path, soi_titles=None, ignore_contains=None):
    """soi_titles / ignore_contains - optional, from the client profile (clients/*.json).
    Settings are local to this call (no global state)."""
    titles = [t.upper() for t in (soi_titles or SOI_TITLES)]
    # profile list may still contain header words (old template) - those use the 2+ header rule
    hw_low = {w.lower() for w in HEADER_WORDS}
    given = list(ignore_contains or IGNORE_CONTAINS)
    header_words = HEADER_WORDS + [x for x in given if x.lower() in hw_low and x not in HEADER_WORDS]
    ignore = [x for x in given if x.lower() not in hw_low]
    rows, suspicious, checks = [], [], []
    acc = {"sum": 0.0, "n": 0, "missing": 0}

    def reset_acc():
        acc.update(sum=0.0, n=0, missing=0)

    def add_acc(fv):
        acc["n"] += 1
        if fv is None:
            acc["missing"] += 1
        else:
            acc["sum"] += fv

    with pdfplumber.open(pdf_path) as pdf:
        fund = section = sector = None
        sector_from_header = False
        in_soi = False
        in_deriv = False
        for pno, page in enumerate(pdf.pages, start=1):
            text = page.extract_text() or ""
            lines = [ln.strip() for ln in text.splitlines()]
            titled = has_title(lines, titles)
            if not titled and not in_soi:
                continue            # not an SOI page - skip the (slow) italics scan
            italics = italic_line_texts(page)
            # "Principal (000)" in the table header: debt holdings are printed in thousands
            head_txt = " ".join(lines[:16])
            debt_mult = 1000 if re.search(r"\(\s*000s?\s*\)", head_txt) else 1
            if titled:
                in_soi, cont = True, False
                new_fund = find_fund(lines, titles) or fund
                if new_fund != fund:  # new fund - sector does not carry over from the previous one
                    fund, section, sector = new_fund, None, None
                    in_deriv = False
                    reset_acc()
            else:
                # Continuation page without the SOI title: accept it only if the previous page was an SOI,
                # the page does not start another statement and it contains security lines.
                head = " ".join(lines[:8]).upper()
                has_secs = any(SEC_RE.match(x) or EQ_RE.match(x) for x in lines)
                if in_soi and has_secs and not any(k in head for k in OTHER_HEADINGS):
                    cont = True
                else:
                    if in_soi and has_secs and not in_deriv:  # securities right after the SOI, but skipped - say so
                        suspicious.append({"Fund": fund, "Page": pno,
                                           "Line": "PAGE SKIPPED: looks like another statement - check it is not an SOI page",
                                           "Buffer": " | ".join(lines[:3])})
                    in_soi = False
                    continue
            buffer = []

            for ln in lines:
                if not ln:
                    continue
                h = HEADER_RE.match(ln)
                if h:
                    label = CONTINUED_RE.sub("", h.group("label").strip())
                    buffer = []
                    if DERIVATIVE_HEADINGS.search(label):
                        in_deriv = True             # forwards / futures / swaps: not part of P&S
                        section, sector = label, None
                        reset_acc()
                        continue
                    in_deriv = False
                    if label == section:
                        continue                    # section heading repeated on the next page - nothing changes
                    # The same sector/section header repeated at the top of the next page while the sector is
                    # still open (no subtotal yet) = the sector CONTINUES: keep the running sum and count,
                    # otherwise the subtotal check only sees the securities from the new page.
                    cont_hdr = acc["n"] > 0 and label == (section if label.isupper() else sector)
                    if label.isupper():
                        section, sector = label, None
                    else:
                        sector, sector_from_header = label, True
                    if not cont_hdr:
                        reset_acc()
                    continue
                if in_deriv:
                    continue
                if ln in italics and not SEC_RE.match(ln) and not EQ_RE.match(ln) and not SUBTOTAL_RE.match(ln):
                    base = CONTINUED_RE.sub("", ln).strip()
                    buffer = []
                    if base != ln.strip() and base == sector:
                        continue                    # 'Banks (continued)' - same sector, keep the running sum
                    # sub-sector heading (italic): the header printed before it becomes the section
                    # (also for the 2nd, 3rd ... section of a fund: 'Corporate Bonds' after 'Asset-Backed Securities')
                    if sector and sector_from_header:
                        section = sector
                        sector_from_header = False
                    sector, sector_from_header = base, False
                    reset_acc()
                    continue

                sub = SUBTOTAL_RE.match(ln)
                if sub and (acc["n"] > 0 or sector or section):
                    if acc["n"] > 0:   # a section total right after the last sector subtotal is not checked
                        pdf_total = to_number(ln.split()[0])
                        diff = round(pdf_total - acc["sum"], 2)
                        checks.append({
                            "Fund": fund, "Page": pno, "Section": section, "Sector": sector,
                            "PDF_Subtotal": pdf_total, "Parsed_Total": acc["sum"], "Difference": diff,
                            "Securities": acc["n"], "Missing_Fair_Value": acc["missing"],
                            "Status": "OK" if abs(diff) < 1 and acc["missing"] == 0 else "DIFFERENCE",
                        })
                    reset_acc()
                    buffer = []
                    continue

                m = SEC_RE.match(ln)
                used = ln
                possible_wrap = ""
                if buffer:
                    if m:
                        # The last line of a wrapped name carries coupon + date, so it matches on its own.
                        # Join the previous line when it looks like the start of a name; otherwise flag it.
                        prev = buffer[-1]
                        mj = SEC_RE.match(prev + " " + ln) if looks_like_name_part(prev) else None
                        if mj:
                            m, used = mj, prev + " " + ln
                        else:
                            possible_wrap = " ".join(buffer)
                    else:
                        joined = " ".join(buffer + [ln])
                        mj = SEC_RE.match(joined)
                        if mj:
                            m, used = mj, joined
                if m:
                    footnotes, holding, fv, pct, desig = split_rest(m.group("rest"))
                    if holding is not None and debt_mult != 1:
                        holding = holding * debt_mult           # "Principal (000)" -> units
                    cpn_txt = m.group("cpn")
                    date_txt = m.group("date")
                    # footnote markers: glued to the coupon "3.526%(1)" or to the end of the name "Class A(2)"
                    cfn = [x.strip() for x in re.findall(r"\(\s*\w{1,3}\s*\)", m.group("cfn") or "")]
                    name_txt, nfn = strip_trailing_footnotes(m.group("name").strip().rstrip(","))
                    footnotes = nfn + cfn + footnotes
                    if parse_date(date_txt) is None and "/" not in date_txt and date_txt.upper() != "PERPETUAL":
                        pub = (name_txt + used[m.end("name"):m.start("cfn")]).strip()   # '-' / 'TBA' not published
                    else:
                        pub = (name_txt + used[m.end("name"):m.start("cfn")] + used[m.end("cfn"):m.end("date")]).strip()
                    rows.append({
                        "Fund": fund,
                        "Page": pno,
                        "Type": "Debt",
                        "Section": section,
                        "Sector": sector,
                        "Issuer": name_txt,
                        "Coupon": None if cpn_txt.upper() in ("ZCP", "TBD") else float(cpn_txt.rstrip("%")),
                        "ZCP": cpn_txt.upper() == "ZCP",
                        "Maturity": parse_date(date_txt),
                        # '..., 08/01/29 144A' - 144A after the date is part of the published name
                        "Name_publication": " ".join([pub] + desig),
                        "Note": date_note(date_txt, cpn_txt),
                        "Footnotes": " ".join(footnotes),
                        "Holding": holding,
                        "Fair_Value": fv,
                        "Pct": pct,
                        "Wrapped": used != ln,
                        "Continuation_Page": cont,
                        "Check_prev_line": possible_wrap,
                        "Raw_line": used,
                    })
                    add_acc(fv)
                    buffer = []
                    continue

                numeric_only = all(NUM_RE.match(t) for t in ln.split())

                # numbers on a separate line below a debt row without numbers -> attach them to it
                if numeric_only and not buffer and rows and rows[-1]["Page"] == pno \
                        and rows[-1]["Type"] == "Debt" and rows[-1]["Fair_Value"] is None:
                    _, h_, fv_, pct_, _ = split_rest(" " + ln)
                    if fv_ is not None:
                        rows[-1].update(Holding=h_, Fair_Value=fv_, Pct=pct_)
                        acc["missing"] -= 1
                        acc["sum"] += fv_
                        continue

                # --- dated securities without a coupon ---
                nc, used_n = NOCPN_RE.match(ln), ln
                if not nc and buffer and looks_like_name_part(buffer[-1]):
                    nj = NOCPN_RE.match(buffer[-1] + " " + ln)
                    if nj:
                        nc, used_n = nj, buffer[-1] + " " + ln
                if nc and not DATE_ANYWHERE_RE.search(nc.group("name")) and "%" not in nc.group("name"):
                    name, nfn = strip_trailing_footnotes(nc.group("name").strip())
                    fvt = nc.group("fv")
                    fn_d = dash_footnote(fvt)
                    hold = to_number(nc.group("hold"))
                    rows.append({
                        "Fund": fund, "Page": pno, "Type": "Debt", "Section": section, "Sector": sector,
                        "Issuer": name, "Coupon": None, "ZCP": False, "Maturity": parse_date(nc.group("date")),
                        "Name_publication": f"{name} {nc.group('date')}",
                        "Note": "NO COUPON",
                        "Footnotes": " ".join(nfn + ([fn_d] if fn_d else [])),
                        "Holding": hold * debt_mult if hold is not None else None,
                        "Fair_Value": fv_number(fvt), "Pct": to_number(nc.group("pct")),
                        "Wrapped": used_n != ln, "Continuation_Page": cont,
                        "Check_prev_line": "", "Raw_line": used_n,
                    })
                    add_acc(fv_number(fvt))
                    buffer = []
                    continue

                # --- equities (no coupon or maturity) ---
                if not is_noise(ln, ignore, header_words) or (buffer and numeric_only):
                    e, used_e = EQ_RE.match(ln), ln
                    if not e and buffer:
                        ej = EQ_RE.match(" ".join(buffer + [ln]))
                        if ej:
                            e, used_e = ej, " ".join(buffer + [ln])
                    if e and (DATE_ANYWHERE_RE.search(e.group("name")) or "%" in e.group("name")):
                        e = None  # a coupon or date in the "name" = a debt line SEC_RE could not read -> To Review
                    if e and (any(k.lower() in e.group("name").lower() for k in ignore)
                              or re.match(r"Total Investments?\b", e.group("name"), re.I)):
                        e = None  # 'Total Investment in Securities (Cost: USD 944,608,028) 956,859,297 ...'
                    if e:
                        name, footnotes = strip_trailing_footnotes(e.group("name").strip())
                        fn_d = dash_footnote(e.group("fv"))
                        if fn_d:
                            footnotes.append(fn_d)
                        rows.append({
                            "Fund": fund, "Page": pno, "Type": "Equity",
                            "Section": section, "Sector": sector,
                            "Issuer": name, "Coupon": None, "ZCP": False, "Maturity": None,
                            "Name_publication": name,
                            "Note": "",
                            "Footnotes": " ".join(footnotes),
                            "Holding": to_number(e.group("hold")),
                            "Fair_Value": fv_number(e.group("fv")),
                            "Pct": to_number(e.group("pct")),
                            "Wrapped": used_e != ln,
                            "Continuation_Page": cont,
                            "Check_prev_line": " ".join(buffer) if (buffer and used_e == ln) else "",
                            "Raw_line": used_e,
                        })
                        add_acc(fv_number(e.group("fv")))
                        buffer = []
                        continue

                if (DATE_ANYWHERE_RE.search(ln) or ln.rstrip(")").endswith("%")) and not is_noise(ln, ignore, header_words):
                    suspicious.append({"Fund": fund, "Page": pno, "Line": ln, "Buffer": " | ".join(buffer)})
                if is_noise(ln, ignore, header_words):
                    buffer = []
                else:
                    buffer = (buffer + [ln])[-(MAX_WRAP_LINES - 1):]
    sus_cols = ["Fund", "Page", "Line", "Buffer"]
    chk_cols = ["Fund", "Page", "Section", "Sector", "PDF_Subtotal", "Parsed_Total", "Difference",
                "Securities", "Missing_Fair_Value", "Status"]
    return pd.DataFrame(rows), pd.DataFrame(suspicious, columns=sus_cols), pd.DataFrame(checks, columns=chk_cols)


def parse_path(path, soi_titles=None, ignore_contains=None, log=print):
    """Accepts a single PDF or a folder of PDFs (e.g. one SOI per fund). Results are combined."""
    p = Path(path)
    files = sorted(f for f in p.glob("*.pdf") if not f.name.startswith("~$")) if p.is_dir() else [p]
    if not files:
        raise ValueError(f"No PDF files found in: {p}")
    parts = []
    for f in files:
        df, sus, chk = parse_pdf(f, soi_titles, ignore_contains)
        for d in (df, sus, chk):
            d.insert(0, "Source_File", f.name)
        if len(files) > 1:
            log(f"  {f.name}: {len(df)} securities, {df['Fund'].nunique() if not df.empty else 0} fund(s)")
        parts.append((df, sus, chk))
    return tuple(pd.concat([pt[i] for pt in parts], ignore_index=True) for i in range(3))


def main():
    if len(sys.argv) > 1:
        pdf_path = Path(sys.argv[1])   # a PDF file or a folder with PDFs
    else:
        import tkinter as tk
        from tkinter import filedialog
        tk.Tk().withdraw()
        p = filedialog.askopenfilename(title="Select the financial statements PDF", filetypes=[("PDF", "*.pdf")])
        if not p:
            return
        pdf_path = Path(p)

    df, sus, chk = parse_path(pdf_path)
    out = (pdf_path / "SOI_parsed.xlsx") if pdf_path.is_dir() else pdf_path.with_name(pdf_path.stem + "_SOI_parsed.xlsx")
    save_excel(df, sus, out, chk)
    bad = int((chk["Status"] != "OK").sum()) if not chk.empty else 0
    print(f"Securities: {len(df)} | To review: {len(sus)} | Funds: {df['Fund'].nunique() if not df.empty else 0}")
    print(f"Sector subtotal check: {len(chk)} checked, {bad} with difference")
    print(f"Saved: {out}")


def save_excel(df, sus, out, chk=None):
    summary = (df.groupby("Fund", dropna=False).size().reset_index(name="Securities")
               if not df.empty else pd.DataFrame(columns=["Fund", "Securities"]))
    with pd.ExcelWriter(out, engine="openpyxl") as xw:
        df.to_excel(xw, sheet_name="SOI", index=False)
        sus.to_excel(xw, sheet_name="To Review", index=False)
        if chk is not None:
            chk.to_excel(xw, sheet_name="Subtotal Check", index=False)
        summary.to_excel(xw, sheet_name="Summary", index=False)
        for ws in xw.book.worksheets:
            for row in ws.iter_rows():  # PDF text starting with '=' must stay text, never a formula
                for c in row:
                    if c.data_type == "f":
                        c.data_type = "s"
            ws.freeze_panes = "A2"
            for col in ws.columns:
                width = max(len(str(c.value)) if c.value is not None else 0 for c in col)
                ws.column_dimensions[col[0].column_letter].width = min(max(10, width + 2), 80)
        ws = xw.book["SOI"]
        for c in ws[1]:  # format the Maturity column by its header, not by a fixed letter
            if c.value == "Maturity":
                for cell in ws[c.column_letter][1:]:
                    cell.number_format = "DD/MM/YYYY"


if __name__ == "__main__":
    main()
